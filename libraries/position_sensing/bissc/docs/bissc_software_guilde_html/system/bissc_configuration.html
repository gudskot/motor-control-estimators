1. Configuration

1.1. System Example Configuration

If you wish to change the default configuration of the system example, edit the example’s header file. The parameters that can be changed are shown below.

Listing 1.1 /solutions/boostxl_posmgr/shared/include/bissc.h
//
// Number of multi-turn and single-turn bits.
// In an application these may be read from the encoder/sensor itself. In this
// demonstration they are hard-coded for simplicity.
//
//*****************************************************************************

//
// Lika EOH36-20-12
//
#define BISSC_SCD_MT_BITS   12U
#define BISSC_SCD_ST_BITS   20U
//

//
// icHaus MHM Evaluation board
//
//#define BISSC_SCD_MT_BITS     16U
//#define BISSC_SCD_ST_BITS     12U


//*****************************************************************************
//
// Specify the BiSS-C clock rate (MA) in MHz
//
//*****************************************************************************

#define BISSC_MA_CLOCK  8.0f // MHz


//*****************************************************************************
//
// For register read/write testing, the address of the bank select register
// is used. This is standard on encoders and safe to read/write from.
//
// The test keeps track of BISSC_READ_COUNT reads in the register_read_data
// array.
//
//*****************************************************************************
#define BISSC_READ_COUNT            8U
#define BISSC_BANK_SELECT           0x40U
#define BISSC_CD_BUSY               1U
#define BISSC_CD_SETUP_COMPLETE     0U
extern uint16_t bisscRegisterReadData[BISSC_READ_COUNT];

//*****************************************************************************
//
// CRC related declarations
//
// The leading 1 on the polynomial (poly) is implied by the size
// The most common polynomials are:
//
// Control Data (CD):
//            x^4 + x + 1 --> 10011 --> 0x13
//                        leading 1 is implied use: Poly = 3, poly bits = 4
//
// Single Cycle Data(SCD):
//            x^6 + x + 1 --> 1000011 --> 0x43
//                        leading 1 is implied use: Poly = 3, poly bits = 6
//
// Specify the CRC starting value. For BiSS-C this is typically 0.
//
//
// Other BiSS polynomials:
//
// x^5 + x^2 + X^0 = 0x25
//                        ---> Poly = 5, poly bits = 5
//
// x^16 + x^15 + x^12 + x^7 + X^6 + x^4 + x^3 + X^0 = 0x190D9
//                        ---> Poly = 0x90D9, poly bits = 16
//
//*****************************************************************************

#define BISSC_CD_CRC_POLY_BITS     4U
#define BISSC_CD_CRC_POLY          0x03U
#define BISSC_CD_CRC_START         0U
#define BISSC_CD_CRCTABLE_SIZE     256U

#define BISSC_SCD_CRC_POLY_BITS    6U
#define BISSC_SCD_CRC_POLY         0x03U
#define BISSC_SCD_CRC_START        0U
#define BISSC_SCD_CRCTABLE_SIZE    256U

//#define BISSC_SCD_CRC_POLY_BITS    16U
//#define BISSC_SCD_CRC_POLY         0x90D9U
//#define BISSC_SCD_CRC_START        0U
//#define BISSC_SCD_CRCTABLE_SIZE    256U

//*****************************************************************************
//
// Delay time, in micro-seconds, between each transaction
// This time does not include the parse of the SCD and CDS.
// In a typical application this time would be controlled by a timer interrupt.
// In this demonstration, a simple while() loop with a delay is used.
//
// The value chosen was based on the read access time of one of the
// tested encoders. This allows the reads to occur even with the limitations
// of the CD state machine implementation. If the state machine is updated
// to allow for the encoder to extend the read, or write, cycle this time
// can be reduced. Other encoders may not require this much time. Refer
// to the encoder specifications.
//
//*****************************************************************************
#define BISSC_IDLE_DELAY_US       200ul

//*****************************************************************************
//
// Power-on to normal operation delay 1.5 seconds. Refer to the specifications
// for a specific encoder.
//
//*****************************************************************************
#define BISSC_POWER_ON_DELAY_US 1500000ul

//

1.2. Library Configuration

If you wish to change the default configuration of the library, edit the library’s public header file. The parameters that can be changed are shown below.

Note

  • If you change the library configuration then recompile both the library and the system example.

Listing 1.2 /libraries/position_sensing/bissc/include/pm_bissc_include.h
//
// #define BISSC_ENCODER_HAS_CD_INTERFACE
//      Some encoders do not support the control-data CD interface.
//      0: the code associated with the control-data frames is not executed.
//      else: the code associated with control-data frames is executed.
//
// #define PM_BISSC_SPI  SPIA_BASE, SPIB_BASE, SPIC_BASE, SPID_BASE
//      Define which SPI peripheral is used.  This MUST match the instance
//      the CLB is clocking and be connected to the RS485 interface logic.
//
//      Note: The CLB output configuration and SPI configuration may need
//      to be modified if this selection is changed.  Refer to the
//      design guide for more information.
//
// #define PM_SPI_FIFO_MAX_LEVEL
//      This is the depth of the SPI FIFO. If you know the message returned
//      from the encoder will always use fewer levels it can be lowered.
//      This parameter is only used when reading the message out of the
//      SPI FIFO and can save cycles.
//
//  #define PM_BISSC_SPI_FIFO_WIDTH
//      Word size used by the SPI peripheral. You shouldn't have to change
//      this definition.
//
#define PM_BISSC_ENCODER_HAS_CD_INTERFACE   1U
#define PM_BISSC_SPI_FIFO_WIDTH            12U
#define PM_BISSC_SPI_FIFO_MAX_LEVEL        16U

//
// SPI selection specified here and the CLB output
// configuration, and the system control setup in the
// communications demo must match.  If one is changed,
// the other must be changed as well.
//
#if defined(_F28P65X) || defined(F28P65x_DEVICE)
    #define PM_BISSC_SPI           SPID_BASE
    #define PM_BISSC_INT_SPI_RX    INT_SPID_RX
    #define PM_BISSC_CLK_GEN_BASE  CLB3_BASE
    #define PM_BISSC_FSM_BASE      CLB4_BASE
#else
    #error Device should match a predefined symbol in the build options.
#endif
